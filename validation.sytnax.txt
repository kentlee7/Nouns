SELECT
    COUNT(*) AS noun_records,
    COUNT(t.token_id) AS matched_token_records
FROM nouns AS n
LEFT JOIN tokens AS t
    ON n.token_id = t.token_id;


-- Check whether the tokens have matching sentence records 

SELECT
    COUNT(*) AS matched_tokens,
    COUNT(s.cum_sent_id) AS matched_sentences
FROM nouns AS n
JOIN tokens AS t
    ON n.token_id = t.token_id
LEFT JOIN sentences AS s
    ON t.cum_sent_id = s.cum_sent_id;

-- Create a reproducible sample of 400 noun tokens.

SET GLOBAL innodb_buffer_pool_size = 1073741824; -- 1 GB in bytes

CREATE TABLE validation_sample AS
SELECT
    n.token_id
FROM nouns AS n
JOIN tokens AS t
    ON n.token_id = t.token_id
JOIN sentences AS s
    ON t.cum_sent_id = s.cum_sent_id
ORDER BY RAND()
LIMIT 400;


-- Verify the sample.

SELECT COUNT(*) AS sample_size
FROM validation_sample;

-- You should get: sample_size  = 400
-- Then check for duplicate token IDs:

SELECT
    token_id,
    COUNT(*) AS records
FROM validation_sample
GROUP BY token_id
HAVING COUNT(*) > 1;

-- This should return no rows.

-- Finally, save the sample IDs somewhere safe. The persistent table will preserve them in the database, but exporting them is also useful for documentation and reproducibility.

SELECT token_id
FROM validation_sample
ORDER BY token_id;


-- For validation annotation 

SELECT
    n.token_id,
    n.token AS noun_token,
    n.lemma,
    t.pos AS original_pos,
    t.token_postn,
    t.sent_id,
    t.cum_sent_id,
    s.text_id,
    s.genre,
    s.year,
    s.pgraph_num,
    s.sent_num,
    st.sentence

FROM validation_sample AS vs

JOIN nouns AS n
    ON vs.token_id = n.token_id

JOIN tokens AS t
    ON n.token_id = t.token_id

JOIN sentences AS s
    ON t.cum_sent_id = s.cum_sent_id

LEFT JOIN (
    SELECT
        cum_sent_id,
        GROUP_CONCAT(
            token
            ORDER BY token_postn
            SEPARATOR ' '
        ) AS sentence
    FROM tokens
    GROUP BY cum_sent_id
) AS st
    ON t.cum_sent_id = st.cum_sent_id

ORDER BY n.token_id;

-- Validation key - how the script classified nouns 
 


SELECT
    n.token_id,
    n.countability AS script_countability,
    n.number AS script_number,
    n.definiteness AS script_definiteness

FROM validation_sample AS vs

JOIN nouns AS n
    ON vs.token_id = n.token_id

ORDER BY n.token_id;

# Identify tokens that can be compared directly.
eligible <- !is.na(validation$ct_manual) &
            !is.na(validation$ct_script) &
            validation$ct_manual %in% valid_labels &
            validation$ct_script %in% valid_labels

v <- validation[eligible, ]

# Confusion matrix: manual annotations in rows,
# automated classifications in columns.
cm <- table(
  Manual = factor(v$ct_manual,
                  levels = valid_labels),
  Script = factor(v$ct_script,
                  levels = valid_labels)
)

print(cm)

# Overall agreement.
agreement <- mean(
  v$ct_manual == v$ct_script
)

cat("Evaluable tokens:", nrow(v), "\n")
cat("Overall agreement:", round(agreement, 4), "\n")
cat("Overall agreement (%):",
    round(100 * agreement, 2), "\n")

# Cohen's kappa for the binary countability labels.
kappa_result <- kappa2(
  data.frame(
    Manual = factor(v$ct_manual,
                    levels = valid_labels),
    Script = factor(v$ct_script,
                    levels = valid_labels)
  ),
  weight = "unweighted"
)

print(kappa_result)
